// eatiseasy-stripe-webhook — Cloudflare Worker // // Qué hace: Stripe le avisa a este Worker cada vez que alguien paga de // verdad (o cancela). El Worker apunta en una "libreta" (Cloudflare KV) // qué emails tienen Pro activo. La app, en vez de fiarse de "?pro=1" en // la URL (que cualquiera puede escribir a mano), le pregunta a este // Worker "¿este email ha pagado de verdad?" — así el "Pro" ya no se // puede falsificar solo escribiendo cosas en la barra de direcciones. // // ============================================================ // CONFIGURACIÓN NECESARIA EN CLOUDFLARE (hazlo en este orden): // ============================================================ // 1. Crea este Worker como uno nuevo (igual que hiciste con el de las // fotos): "Create" → "Worker" (NO "Connect to Git") → pega este // código entero → "Deploy". // // 2. Crea el almacén KV: en el panel de Cloudflare, ve a // "Workers & Pages" → "KV" (en el menú lateral) → "Create a // namespace" → llámalo, por ejemplo, "eatiseasy-pro-store". // // 3. Conecta ese KV a este Worker: entra en este Worker → "Settings" → // "Variables and Secrets" (o "Bindings", según la versión del // panel) → añade un "KV Namespace Binding": // - Variable name: PRO_STORE (exactamente así, en mayúsculas) // - KV namespace: el que acabas de crear // Guarda. // // 4. Añade el secret STRIPE_WEBHOOK_SECRET (lo consigues en el paso 5): // "Settings" → "Variables and Secrets" → "Add" → nombre // STRIPE_WEBHOOK_SECRET → tipo Secret. // // 5. En Stripe: dashboard.stripe.com → "Developers" → "Webhooks" → // "Add endpoint". URL del endpoint: la URL de este Worker + "/webhook" // (ej. https://eatiseasy-stripe.tuusuario.workers.dev/webhook). // Eventos a escuchar: marca "checkout.session.completed" (el pago) y // "charge.refunded" (por si alguna vez reembolsas una compra — con // pago único es la ÚNICA forma de que alguien pierda el acceso, no // hay "renovación" que pueda fallar). Al crearlo, Stripe te da un // "Signing secret" (empieza por whsec_) — cópialo y pégalo como el // secret STRIPE_WEBHOOK_SECRET del paso 4. // // 6. Añade OTRO secret: STRIPE_SECRET_KEY. Se consigue en Stripe → // "Developers" → "API keys" (modo Live) → "Secret key" (empieza por // sk_live_...). OJO: este es distinto al Signing secret del paso 5, // y es más sensible — nunca lo pegues en ningún sitio público. // // 7. Pega la URL de este Worker (+ "/check-pro") en el código de la // app, en la constante PRO_VERIFY_ENDPOINT. // // Cambia esto por tu dominio real antes de publicar: const ALLOWED_ORIGIN = 'https://ikernebreda9-dot.github.io'; export default { async fetch(request, env) { const url = new URL(request.url); if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders() }); } // Stripe llama aquí cuando pasa algo real (pago hecho, cancelado...) if (url.pathname === '/webhook' && request.method === 'POST') { return handleStripeWebhook(request, env); } // La app llama aquí justo al volver del pago, con el ID de la sesión // de Stripe — le pregunta a Stripe directamente si esa compra // concreta se pagó de verdad, sin depender de que ningún email // coincida (soluciona el lío de Apple Pay / Link usando un email // distinto al de la cuenta de Google). if (url.pathname === '/verify-session' && request.method === 'GET') { return handleVerifySession(url, env); } // La app llama aquí para comprobar si un email tiene Pro de verdad if (url.pathname === '/check-pro' && request.method === 'GET') { return handleCheckPro(url, env); } return new Response('Not found', { status: 404, headers: corsHeaders() }); } }; async function handleStripeWebhook(request, env) { const payload = await request.text(); const sigHeader = request.headers.get('stripe-signature'); console.log('[webhook] petición recibida. ¿Trae cabecera stripe-signature?', !!sigHeader); if (!env.STRIPE_WEBHOOK_SECRET) { console.log('[webhook] ERROR: falta el secret STRIPE_WEBHOOK_SECRET'); return new Response('Falta configurar el secret STRIPE_WEBHOOK_SECRET en este Worker.', { status: 500 }); } if (!env.PRO_STORE) { console.log('[webhook] ERROR: falta el KV namespace PRO_STORE'); return new Response('Falta conectar el KV namespace PRO_STORE a este Worker.', { status: 500 }); } console.log('[webhook] secret y KV están presentes, primeros 6 caracteres del secret configurado:', env.STRIPE_WEBHOOK_SECRET.slice(0, 6)); const valid = await verifyStripeSignature(payload, sigHeader, env.STRIPE_WEBHOOK_SECRET); console.log('[webhook] ¿firma válida?', valid); if (!valid) { return new Response('Firma inválida', { status: 400 }); } let event; try { event = JSON.parse(payload); } catch (e) { console.log('[webhook] ERROR: JSON inválido', e.message); return new Response('JSON inválido', { status: 400 }); } console.log('[webhook] tipo de evento recibido:', event.type); // Pago único completado: Pro para siempre (sin fecha de caducidad). // La única forma de perder el acceso a partir de aquí es un reembolso // real (evento charge.refunded, más abajo). if (event.type === 'checkout.session.completed') { const session = event.data.object; let email = (session.customer_details && session.customer_details.email) || session.customer_email || null; const customerId = session.customer; console.log('[webhook] checkout.session.completed | email:', email, '| customerId:', customerId); if (email) { email = email.toLowerCase().trim(); await env.PRO_STORE.put('pro:' + email, JSON.stringify({ active: true, expiresAt: null, updatedAt: Date.now() })); console.log('[webhook] guardado: pro:' + email + ' activo para siempre'); if (customerId) { await env.PRO_STORE.put('customer:' + customerId, email); } } else { console.log('[webhook] AVISO: sin email, no se ha guardado nada'); } } // Reembolso: le quitamos el Pro. Es la única vía de revocación con // pago único (no hay suscripción que cancelar ni renovación que falle). if (event.type === 'charge.refunded') { const charge = event.data.object; let email = (charge.billing_details && charge.billing_details.email) || null; const customerId = charge.customer; if (!email && customerId) { email = await env.PRO_STORE.get('customer:' + customerId); } if (email) { email = email.toLowerCase().trim(); await env.PRO_STORE.put('pro:' + email, JSON.stringify({ active: false, expiresAt: 0, updatedAt: Date.now() })); console.log('[webhook] reembolsada la compra de', email, '— Pro revocado'); } else { console.log('[webhook] AVISO: reembolso sin email identificable, no se ha revocado nada'); } } return new Response('ok', { status: 200 }); } async function handleCheckPro(url, env) { const email = (url.searchParams.get('email') || '').toLowerCase().trim(); if (!email || !env.PRO_STORE) { return new Response(JSON.stringify({ isPro: false }), { headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } const raw = await env.PRO_STORE.get('pro:' + email); // Con pago único, "activo" basta — no hay fecha de caducidad que mirar. const isPro = raw ? !!JSON.parse(raw).active : false; return new Response(JSON.stringify({ isPro }), { headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } async function handleVerifySession(url, env) { const sessionId = url.searchParams.get('session_id'); if (!sessionId) { return new Response(JSON.stringify({ paid: false, error: 'falta session_id' }), { headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } if (!env.STRIPE_SECRET_KEY) { console.log('[verify-session] ERROR: falta el secret STRIPE_SECRET_KEY'); return new Response(JSON.stringify({ paid: false, error: 'Worker sin configurar (falta STRIPE_SECRET_KEY)' }), { status: 500, headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } try { const resp = await fetch(`https://api.stripe.com/v1/checkout/sessions/${encodeURIComponent(sessionId)}`, { headers: { 'Authorization': 'Bearer ' + env.STRIPE_SECRET_KEY } }); const session = await resp.json(); if (!resp.ok) { console.log('[verify-session] Stripe respondió con error:', JSON.stringify(session).slice(0, 200)); return new Response(JSON.stringify({ paid: false, error: 'Stripe no reconoce esa sesión' }), { headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } const paid = session.payment_status === 'paid' || session.status === 'complete'; const email = (session.customer_details && session.customer_details.email) || session.customer_email || null; console.log('[verify-session] sesión', sessionId, '| pagada:', paid, '| email:', email); if (paid && email && env.PRO_STORE) { await env.PRO_STORE.put('pro:' + email.toLowerCase().trim(), JSON.stringify({ active: true, since: Date.now() })); } return new Response(JSON.stringify({ paid, email }), { headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } catch (e) { console.log('[verify-session] ERROR inesperado:', e.message); return new Response(JSON.stringify({ paid: false, error: e.message }), { status: 500, headers: { ...corsHeaders(), 'Content-Type': 'application/json' } }); } } // Comprueba que el aviso viene realmente de Stripe (y no de cualquiera // que le mande una petición falsa a este Worker) usando la firma que // Stripe añade en la cabecera "stripe-signature". async function verifyStripeSignature(payload, sigHeader, secret) { if (!sigHeader) return false; const parts = Object.fromEntries(sigHeader.split(',').map(p => p.split('='))); const timestamp = parts['t']; const signature = parts['v1']; if (!timestamp || !signature) return false; const signedPayload = `${timestamp}.${payload}`; const enc = new TextEncoder(); const key = await crypto.subtle.importKey('raw', enc.encode(secret), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']); const sigBuffer = await crypto.subtle.sign('HMAC', key, enc.encode(signedPayload)); const expected = Array.from(new Uint8Array(sigBuffer)).map(b => b.toString(16).padStart(2, '0')).join(''); return expected === signature; } function corsHeaders() { return { 'Access-Control-Allow-Origin': ALLOWED_ORIGIN, 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type, stripe-signature' }; }
